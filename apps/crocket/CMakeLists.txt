cmake_minimum_required(VERSION 3.28)
project(bench_crocket CXX)
include(FetchContent)
set(CROCKET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CROCKET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CROCKET_PG ON CACHE STRING "" FORCE)
FetchContent_Declare(crocket GIT_REPOSITORY https://github.com/sean3z/crocket.git GIT_TAG 0.5.2)
FetchContent_MakeAvailable(crocket)
add_executable(bench main.cpp)
target_link_libraries(bench PRIVATE crocket::crocket crocket::pg)
